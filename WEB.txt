// GUARDAR FAVORITOS EN ESTE ORDENADOR

let ligaActual = "";
let favoritos = JSON.parse(localStorage.getItem("goalnowFavoritos")) || [];

function guardarFavoritos() {    localStorage.setItem(
        "goalnowFavoritos",
        JSON.stringify(favoritos)
    );
}


// PANTALLA DE INICIO

function inicio() {
    let html = `
        <h2>Todos los deportes</h2>
        <p>Elige un deporte para descubrir sus competiciones.</p>
    `;

    for (let deporte in deportes) {
        html += `
            <button class="sport-button" onclick="mostrarLigas('${deporte}')">
                ${deporte}
            </button>
        `;
    }

    document.getElementById("contenido").innerHTML = html;
}


// MOSTRAR LAS LIGAS DE UN DEPORTE

function mostrarLigas(deporte) {
    let html = `
        <h2>${deporte}</h2>
        <p>Elige una competición:</p>
    `;

    deportes[deporte].forEach(liga => {
        html += `
            <div class="panel">
                <button onclick="mostrarClasificacion('${liga}')">
                    ${liga} →
                </button>
            </div>
        `;
    });

    html += `<button onclick="inicio()">Volver</button>`;

    document.getElementById("contenido").innerHTML = html;
}


// MOSTRAR CLASIFICACIÓN

 function mostrarClasificacion(liga) {
    ligaActual = liga;

    let lista = equipos[liga] || [
        "Participante 1",
        "Participante 2",
        "Participante 3",
        "Participante 4"
    ];

    let html = `
        <h2>Clasificación: ${liga}</h2>
        <p>Temporada 2026-2027</p>
        <p class="small">
            Tabla de demostración. No son posiciones ni puntos reales.
        </p>

        <div class="panel">
        <table>
            <tr>
                <th>Pos.</th>
                <th>Equipo</th>
                <th>Puntos</th>
                <th>⭐</th>
            </tr>
    `;

    lista.forEach((equipo, i) => {
        let marcado = favoritos.includes(equipo);

        html += `
            <tr>
                <td>${i + 1}</td>
                <td>${equipo}</td>
                <td>—</td>
                <td>
                    <button class="${marcado ? "favorite" : ""}"
                    onclick="cambiarFavorito('${equipo}')">
                        ${marcado ? "★" : "☆"}
                    </button>
                </td>
            </tr>
        `;
    });

    html += `
        </table>
        </div>
        <button onclick="verFavoritos()">Ver favoritos</button>
        <button onclick="inicio()">Inicio</button>
    `;

    document.getElementById("contenido").innerHTML = html;
}

// AÑADIR O QUITAR UN EQUIPO FAVORITO

function cambiarFavorito(equipo) {
    if (favoritos.includes(equipo)) {
        favoritos = favoritos.filter(e => e !== equipo);
    } else {
        favoritos.push(equipo);
    }

    guardarFavoritos();
    mostrarClasificacion(ligaActual);
}

// PESTAÑA DE FAVORITOS

function verFavoritos() {
    let html = `
        <h2>⭐ Mis favoritos</h2>
        <p>Estos son tus equipos seleccionados:</p>
    `;

    if (favoritos.length === 0) {
        html += `
            <div class="panel">
                Todavía no tienes equipos favoritos.
                Entra en una clasificación y pulsa ☆.
            </div>
        `;
    }

    favoritos.forEach(equipo => {
        html += `
            <div class="panel">
                <h3>⭐ ${equipo}</h3>
                <p>Partidos de ${equipo}</p>
                <p class="small">
                    Los partidos reales aparecerán cuando conectemos
                    GoalNow a una fuente de datos deportivos.
                </p>
                <button onclick="quitarFavorito('${equipo}')">
                    Quitar de favoritos
                </button>
            </div>
        `;
    });

    html += `<button onclick="inicio()">Volver al inicio</button>`;

    document.getElementById("contenido").innerHTML = html;
}


function quitarFavorito(equipo) {
    favoritos = favoritos.filter(e => e !== equipo);
    guardarFavoritos();
    verFavoritos();
}
